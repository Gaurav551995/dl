import numpy as np
import tensorflow as tf
from tensorflow.keras import layers, Model
import matplotlib.pyplot as plt

# a. Load & Subset CIFAR-10
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()
x_train, y_train = x_train[:10000], y_train[:10000]
x_test, y_test = x_test[:2000], y_test[:2000]

class_names = ["airplane", "automobile", "bird", "cat", "deer",
               "dog", "frog", "horse", "ship", "truck"]

# b. Preprocessing (Resize to 96x96 and scale to [-1, 1])
def preprocess(images):
    images = tf.image.resize(images, (96, 96))
    return tf.keras.applications.mobilenet_v2.preprocess_input(tf.cast(images, tf.float32))

x_train = preprocess(x_train)
x_test = preprocess(x_test)

# c. Load MobileNetV2 Base & Freeze
base_model = tf.keras.applications.MobileNetV2(
    input_shape=(96, 96, 3),
    include_top=False,
    weights="imagenet"
)
base_model.trainable = False

# d. Custom Classification Head
inputs = layers.Input(shape=(96, 96, 3))
x = base_model(inputs, training=False)
x = layers.GlobalAveragePooling2D()(x)
x = layers.Dense(128, activation="relu")(x)
x = layers.Dropout(0.3)(x)
outputs = layers.Dense(10, activation="softmax")(x)

model = Model(inputs, outputs)

# Stage 1: Train Head Only (lr=0.001)
model.compile(optimizer=tf.keras.optimizers.Adam(0.001),
              loss="sparse_categorical_crossentropy", metrics=["accuracy"])
history_frozen = model.fit(x_train, y_train, epochs=3, batch_size=32, validation_split=0.1)

# Stage 2: Fine-Tune Top 20 Layers (lr=0.0001)
base_model.trainable = True
for layer in base_model.layers[:-20]:
    layer.trainable = False

model.compile(optimizer=tf.keras.optimizers.Adam(0.0001),
              loss="sparse_categorical_crossentropy", metrics=["accuracy"])
history_finetune = model.fit(x_train, y_train, epochs=2, batch_size=32, validation_split=0.1)

# e. Evaluation
loss, accuracy = model.evaluate(x_test, y_test)
print(f"Test Loss: {loss}, Test Accuracy: {accuracy * 100:.2f}%")

# f. Qualitative Sample Prediction Grid (3x4)
predictions = model.predict(x_test[:12], verbose=0)
pred_labels = np.argmax(predictions, axis=1)

fig, axes = plt.subplots(3, 4, figsize=(10, 8))
for i, ax in enumerate(axes.flat):
    image = (x_test[i].numpy() + 1.0) / 2.0  # Invert [-1, 1] to [0, 1] for display
    ax.imshow(np.clip(image, 0, 1))
    p_name, t_name = class_names[pred_labels[i]], class_names[int(y_test[i][0])]
    ax.set_title(f"P: {p_name}\nT: {t_name}", color="green" if p_name == t_name else "red")
    ax.axis("off")
plt.tight_layout(); plt.show()
